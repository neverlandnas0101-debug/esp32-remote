// ========================================== // 🌟 真實雷達動畫引擎 (Canvas 動態渲染) // ========================================== const canvas = document.getElementById('radarCanvas'); const ctx = canvas.getContext('2d'); const radarRadius = 130; const centerX = canvas.width / 2; const centerY = canvas.height - 10; let targetAngle = 90; let displayAngle = 0; // 🌟 讓指針從 0 度開始 let scanDirection = 1; // 🌟 控制掃描方向 (1: 往右, -1: 往左) let obstacleMemory = {}; function renderRadarLoop() { // 1. 清除舊畫布 ctx.clearRect(0, 0, canvas.width, canvas.height); // 2. 畫雷達底色 ctx.beginPath(); ctx.arc(centerX, centerY, radarRadius, Math.PI, 0); ctx.fillStyle = 'rgba(0, 30, 0, 0.8)'; ctx.fill(); // 3. 畫同心圓與角度網格線 ctx.strokeStyle = 'rgba(0, 180, 0, 0.5)'; ctx.lineWidth = 1; for(let i=1; i<=3; i++) { ctx.beginPath(); ctx.arc(centerX, centerY, radarRadius * (i/3), Math.PI, 0); ctx.stroke(); } [0, 30, 90, 150, 180].forEach(deg => { let rad = (180 - deg) * Math.PI / 180; ctx.beginPath(); ctx.moveTo(centerX, centerY); ctx.lineTo(centerX + Math.cos(rad) * radarRadius, centerY - Math.sin(rad) * radarRadius); ctx.stroke(); }); // 4. 🌟 動態指針動畫 (自動循環來回掃描) displayAngle += scanDirection * 1.5; // 這裡的 1.5 是速度，數字越大掃越快 if (displayAngle >= 180) { displayAngle = 180; scanDirection = -1; } if (displayAngle <= 0) { displayAngle = 0; scanDirection = 1; } let currentRad = (180 - displayAngle) * Math.PI / 180; // 🌟 畫出超酷的雷達漸層殘影 ctx.beginPath(); ctx.moveTo(centerX, centerY); ctx.arc(centerX, centerY, radarRadius, currentRad, currentRad + (scanDirection * 0.35), scanDirection === -1); ctx.lineTo(centerX, centerY); ctx.fillStyle = 'rgba(0, 255, 0, 0.25)'; ctx.fill(); // 畫出目前的掃描指針 (亮綠色線) ctx.beginPath(); ctx.moveTo(centerX, centerY); ctx.lineTo(centerX + Math.cos(currentRad) * radarRadius, centerY - Math.sin(currentRad) * radarRadius); ctx.strokeStyle = '#00ff00'; ctx.lineWidth = 2; ctx.stroke(); // 5. 畫出障礙物點 (支援多點與漸隱) let now = Date.now(); let activeWarnings = []; for (let ang in obstacleMemory) { let b = obstacleMemory[ang]; let age = now - b.time; // 如果這個點超過 2.5 秒沒更新，就讓它慢慢淡出消失 if (age < 2500) { let opacity = 1.0 - (age / 2500); let r = (180 - parseInt(ang)) * Math.PI / 180; let distRatio = Math.min(b.dist / 100, 1); const pointX = centerX + Math.cos(r) * radarRadius * distRatio; const pointY = centerY - Math.sin(r) * radarRadius * distRatio; ctx.beginPath(); ctx.arc(pointX, pointY, 6, 0, Math.PI*2); ctx.fillStyle = `rgba(255, ${parseInt(ang) === 90 ? 0 : 100}, 0, ${opacity})`; ctx.fill(); let dirStr = parseInt(ang) === 90 ? "前方" : (parseInt(ang) === 150 ? "左方" : "右方"); activeWarnings.push(`${dirStr}${b.dist}cm`); } } // 6. 更新文字狀態 const distEl = document.getElementById('radarDist'); if (activeWarnings.length > 0) { distEl.innerText = "⚠️ 障礙: " + activeWarnings.join(" | "); distEl.style.color = '#ff4d4d'; } else { distEl.innerText = "✅ 周圍安全 (掃描中...)"; distEl.style.color = '#00ff00'; } // 不斷重複繪製 requestAnimationFrame(renderRadarLoop); }
